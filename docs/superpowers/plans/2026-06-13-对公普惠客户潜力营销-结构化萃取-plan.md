> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 按“科技型企业普惠贷款营销访谈提纲模板.xlsx”的“营销”sheet 表头，构建 12 条结构化知识条目，重新跑通 Step3→Step5，并整理成完整 Agent Skill 包。

    OUT_DIR.mkdir(parents=True, exist_ok=True)
    wb = openpyxl.Workbook()
    ws = wb.active
    ws.title = "营销"

    headers = [
        "场景", "场景说明", "子场景", "子场景说明",
        "步骤", "具体方法", "知识引用", "规则引用",
        "专业术语", "关键输出-名称", "关键输出-描述",
    ]
    ws.append(headers)

    header_font = Font(bold=True)
    for col_idx, _ in enumerate(headers, 1):
        ws.cell(row=1, column=col_idx).font = header_font

    thin = Side(style="thin", color="CCCCCC")
    for row in ROWS:
        ws.append([
            SCENARIO,
            SCENARIO_DESC,
            SUB_SCENARIO,
            SUB_SCENARIO_DESC,
            row["步骤"],
            row["具体方法"],
            row["知识引用"],
            row["规则引用"],
            row["专业术语"],
            row["关键输出-名称"],
            row["关键输出-描述"],
        ])

    for row in ws.iter_rows():
        for cell in row:
            cell.border = Border(left=thin, right=thin, top=thin, bottom=thin)
            cell.alignment = Alignment(wrap_text=True, vertical="top")

    ws.freeze_panes = "A2"
    for col_idx in range(1, len(headers) + 1):
        ws.column_dimensions[openpyxl.utils.get_column_letter(col_idx)].width = 28

    wb.save(OUT_PATH)
    print(f"[OK] 已生成: {OUT_PATH}")
    return OUT_PATH


if __name__ == "__main__":

def create_pipeline() -> str:

def run_step3_finalize(pipeline_id: str) -> None:
    r = requests.post(
        BASE + "/api/step3/finalize",

def run_step4_compile(pipeline_id: str) -> None:
    r = requests.post(

    pid = create_pipeline()
    seed_step2_file(pid)
    run_step3_finalize(pid)

    run_step5_replay(pid)
